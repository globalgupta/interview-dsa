1. Two Sum — O(n²) 

let nums = [2,7,11,15]
let target = 9

for(let i=0; i<nums.length; i++){
    for(let j=i+1; j<nums.length; j++){
        if(nums[i] + nums[j] === target){
            console.log(i, j)
        }
    }
}
2. Valid Parentheses — Stack 

let str = "[{()}]"
let stack = []

for(let i=0; i<str.length; i++){
    let ch = str[i]

    if(ch == "[" || ch == "{" || ch == "("){
        stack.push(ch)
    }
    else{
        let last = stack.pop()

        if(
        (ch == "]" && last != "[") ||
        (ch == "}" && last != "{") ||
        (ch == ")" && last != "(")
        ){
            console.log("Invalid")
            return
        }
    }
}

if(stack.length == 0)
    console.log("Valid")
else
    console.log("Invalid")

    let ch = str[i]
    
    if(obj[ch]) {
        obj[ch] = obj[ch] + 1

    }
}

4. Flatten Nested Array — Recursion 

let arr = [1, [2, [3, 4], 5]]

function flattenArr(arr) {
 let result = []

 for(let i=0; i<arr.length; i++) {
    let val = arr[i]

    if(Array.isArray(val)) {
        result = result.concat(flattenArr(val))
    }
    else {
        result.push(val)
    }
 }

 return result
}

console.log(flattenArr(arr))
5. Print 1 to 10 with 1s Delay 

// Using setInterval
let i = 1;

let interval = setInterval(()=> {
    console.log(i)
    i++
    
    if(i > 10) {
        clearInterval(interval)
    }
}, 1000)


// Using setTimeout + closure
for(let i=1; i<=10; i++) {
    (function(x) {
        setTimeout(()=> {
            console.log(x)
        }, x*1000)
    })(i)
}
6. Second Highest in Array — O(n) 

let arr = [5, 8, 9, 11, 4, 2, 1, 15, 13]

let first = 0
let second = 0

for(let i=0; i<arr.length; i++) {
  if(arr[i] > first) {
    second = first
    first = arr[i]
  }

  if(arr[i] > second && arr[i] !== first) {
    second = arr[i]
  }
}

console.log(first, second)
